import { CandidateCase, CaseProfile, ScoreBreakdown } from './types';

export interface RankedCandidate {
  candidate: CandidateCase;
  score: number;
  score_breakdown: ScoreBreakdown;
}

function normalizeTokens(values: string[]): string[] {
  return values
    .join(' ')
    .toLowerCase()
    .replace(/[^a-z0-9\s]/g, ' ')
    .split(/\s+/)
    .filter(token => token.length > 2);
}

function countOverlap(needles: string[], haystack: string): number {
  const text = haystack.toLowerCase();
  const uniqueNeedles = Array.from(new Set(needles));
  return uniqueNeedles.filter(token => text.includes(token)).length;
}

function scoreTextOverlap(maxScore: number, needles: string[], haystack: string): number {
  if (needles.length === 0) return 0;
  const overlap = countOverlap(needles, haystack);
  return Math.min(maxScore, Math.round((overlap / Math.min(needles.length, 12)) * maxScore));
}

function scoreCourt(candidate: CandidateCase): number {
  const court = candidate.court.toLowerCase();
  if (court.includes('supreme court')) return 8;
  if (court.includes('high court')) return 7;
  return 0;
}

function scoreRecency(candidate: CandidateCase): number {
  const year = Number(candidate.date.match(/\d{4}/)?.[0]);
  if (!year) return 0;
  if (year >= 2020) return 5;
  if (year >= 2010) return 3;
  return 1;
}

export function rankCandidates(
  profile: CaseProfile,
  candidates: CandidateCase[],
  limit = 20
): RankedCandidate[] {
  const legalIssueTokens = normalizeTokens(profile.legal_issues);
  const factTokens = normalizeTokens(profile.facts);
  const statuteTokens = normalizeTokens(profile.statutes);
  const keywordTokens = normalizeTokens(profile.keywords);

  return candidates
    .map(candidate => {
      const searchableText = [
        candidate.title,
        candidate.court,
        candidate.snippet,
        candidate.raw_text || '',
      ].join(' ');

      const score_breakdown: ScoreBreakdown = {
        legal_issue: scoreTextOverlap(40, [...legalIssueTokens, ...keywordTokens], searchableText),
        facts: scoreTextOverlap(25, factTokens, searchableText),
        statutes: scoreTextOverlap(20, statuteTokens, searchableText),
        court: scoreCourt(candidate),
        recency: scoreRecency(candidate),
      };

      const score = Math.min(
        100,
        score_breakdown.legal_issue +
          score_breakdown.facts +
          score_breakdown.statutes +
          score_breakdown.court +
          score_breakdown.recency
      );

      return { candidate, score, score_breakdown };
    })
    .sort((a, b) => b.score - a.score)
    .slice(0, limit);
}
```

import { CandidateCase, SearchQuery } from '../types';
import { SourceConnector } from './types';

interface IndianKanoonConnectorOptions {
  token?: string;
  baseUrl?: string;
  publicBaseUrl?: string;
  fetcher?: typeof fetch;
}

function stripHtml(value: string): string {
  return value
    .replace(/<[^>]*>/g, ' ')
    .replace(/\s+/g, ' ')
    .trim();
}

export class IndianKanoonConnector implements SourceConnector {
  name = 'Indian Kanoon';
  private token?: string;
  private baseUrl: string;
  private publicBaseUrl: string;
  private fetcher: typeof fetch;

  constructor(options: IndianKanoonConnectorOptions = {}) {
    this.token = options.token || process.env.INDIAN_KANOON_API_TOKEN;
    this.baseUrl = options.baseUrl || process.env.INDIAN_KANOON_API_BASE_URL || 'https://api.indiankanoon.org';
    this.publicBaseUrl = options.publicBaseUrl || 'https://indiankanoon.org';
    this.fetcher = options.fetcher || fetch;
  }

  async search(searchQuery: SearchQuery): Promise<CandidateCase[]> {
    if (!this.token) {
      throw new Error('INDIAN_KANOON_API_TOKEN is required for public precedent search');
    }

    const formInput = searchQuery.court_filter
      ? `${searchQuery.query} doctypes:${searchQuery.court_filter}`
      : searchQuery.query;

    const url = new URL('/search/', this.baseUrl);
    url.searchParams.set('formInput', formInput);
    url.searchParams.set('pagenum', '0');

    const response = await this.fetcher(url.toString(), {
      headers: {
        Authorization: `Token ${this.token}`,
        Accept: 'application/json',
      },
    });

    if (!response.ok) {
      throw new Error(`Indian Kanoon search failed with status ${response.status}`);
    }

    const json = await response.json();
    const docs = Array.isArray(json.docs) ? json.docs : [];

    return docs.map((doc: any): CandidateCase => {
      const externalId = String(doc.tid || doc.docid || '');
      return {
        id: `indiankanoon:${externalId}`,
        source: this.name,
        external_id: externalId,
        title: stripHtml(String(doc.title || 'Untitled case')),
        court: stripHtml(String(doc.docsource || '')),
        date: stripHtml(String(doc.publishdate || doc.date || '')),
        url: `${this.publicBaseUrl}/doc/${externalId}/`,
        snippet: stripHtml(String(doc.headline || '')),
        doc_size: typeof doc.docsize === 'number' ? doc.docsize : undefined,
      };
    }).filter((candidate: CandidateCase) => candidate.external_id.length > 0);
  }

  async fetchDocumentText(externalId: string): Promise<string> {
    if (!this.token) {
      throw new Error('INDIAN_KANOON_API_TOKEN is required for document fetch');
    }

    const response = await this.fetcher(`${this.baseUrl}/doc/${externalId}/`, {
      headers: {
        Authorization: `Token ${this.token}`,
        Accept: 'application/json',
      },
    });

    if (!response.ok) {
      throw new Error(`Indian Kanoon document fetch failed with status ${response.status}`);
    }

    const json = await response.json();
    return stripHtml(String(json.doc || json.text || ''));
  }
}
```

import { CaseProfile, CaseProfileSchema, CandidateCase } from './types';

  SEARCH_QUERY_PROMPT,

interface MatchExplanation {

  differences: string[];
  important_points: string[];
  reusable_arguments: string[];
  confidence: 'high' | 'medium' | 'low';
}

async function completeJson(systemPrompt: string, userContent: string, maxTokens = 1800): Promise<any> {
  const completion = await openai.chat.completions.create({
    model: 'gpt-4o-mini',
    messages: [
      { role: 'system', content: systemPrompt },
      { role: 'user', content: userContent },
    ],
    temperature: 0.1,
    max_tokens: maxTokens,
    response_format: { type: 'json_object' },
  });

  const content = completion.choices[0]?.message?.content;
  if (!content) {
    throw new Error('AI response was empty');
  }

  return JSON.parse(content);
}

export async function extractCaseProfile(documentText: string): Promise<CaseProfile> {
  const parsed = await completeJson(CASE_PROFILE_PROMPT, documentText, 2200);
  return CaseProfileSchema.parse(parsed);
}

export async function generateSearchQueries(profile: CaseProfile): Promise<string[]> {
  const parsed = await completeJson(SEARCH_QUERY_PROMPT, JSON.stringify(profile), 1000);
  return Array.isArray(parsed.queries)
    ? parsed.queries.map(String).filter(Boolean).slice(0, 5)
    : [];
}

export async function explainMatch(profile: CaseProfile, candidate: CandidateCase): Promise<MatchExplanation> {
  const parsed = await completeJson(
    MATCH_EXPLANATION_PROMPT,
    JSON.stringify({
      uploaded_case_profile: profile,
      candidate_case: {
        title: candidate.title,
        court: candidate.court,
        date: candidate.date,
        snippet: candidate.snippet,
        text: candidate.raw_text || '',
      },
    }),
    1600
  );

  return {
    matching_points: Array.isArray(parsed.matching_points) ? parsed.matching_points.map(String) : [],
    differences: Array.isArray(parsed.differences) ? parsed.differences.map(String) : [],
    important_points: Array.isArray(parsed.important_points) ? parsed.important_points.map(String) : [],
    reusable_arguments: Array.isArray(parsed.reusable_arguments) ? parsed.reusable_arguments.map(String) : [],
    confidence: ['high', 'medium', 'low'].includes(parsed.confidence) ? parsed.confidence : 'low',
  };
}

import { extractCaseProfile, explainMatch, generateSearchQueries } from './ai';
import { rankCandidates } from './ranking';
import {
  CandidateCase,
  ResearchMatch,
  ResearchResult,
  SearchQuery,
} from './types';
import { SourceConnector } from './sources/types';

interface ResearchDocument {
  name: string;
  text: string;
}

interface RunPrecedentResearchOptions {
  documents: ResearchDocument[];
  connectors: SourceConnector[];
  now?: () => Date;
  topCandidateLimit?: number;
  deepComparisonLimit?: number;
}

function uniqueCandidates(candidates: CandidateCase[]): CandidateCase[] {
  const seen = new Set<string>();
  return candidates.filter(candidate => {
    if (seen.has(candidate.id)) return false;
    seen.add(candidate.id);
    return true;
  });
}

function buildDocumentText(documents: ResearchDocument[]): string {
  return documents
    .map(document => `DOCUMENT: ${document.name}\n${document.text}`)
    .join('\n\n---\n\n');
}

function attributionFor(source: string): string {
  if (source.toLowerCase().includes('indian kanoon')) return 'Powered by IKanoon';
  return `Source: ${source}`;
}

export async function runPrecedentResearch({
  documents,
  connectors,
  now = () => new Date(),
  topCandidateLimit = 30,
  deepComparisonLimit = 15,
}: RunPrecedentResearchOptions): Promise<ResearchResult> {
  const combinedText = buildDocumentText(documents);
  const caseProfile = await extractCaseProfile(combinedText);
  const generatedQueries = await generateSearchQueries(caseProfile);

  const queries: SearchQuery[] = generatedQueries.map(query => ({
    query,
    source: 'Indian Kanoon',
    court_filter: 'supremecourt,highcourts',
  }));

  const allCandidates: CandidateCase[] = [];
  const warnings = ['Results are based on connected public sources and may not include every relevant precedent.'];

  for (const connector of connectors) {
    for (const query of queries) {
      try {
        const found = await connector.search(query);
        allCandidates.push(...found);
      } catch (error: any) {
        warnings.push(`${connector.name} search failed for one query.`);
      }
    }
  }

  const ranked = rankCandidates(caseProfile, uniqueCandidates(allCandidates), topCandidateLimit);
  const selected = ranked.slice(0, deepComparisonLimit);
  const matches: ResearchMatch[] = [];

  for (const rankedCandidate of selected) {
    const candidate = { ...rankedCandidate.candidate };

    try {
      candidate.raw_text = await connectors
        .find(connector => connector.name === candidate.source)
        ?.fetchDocumentText(candidate.external_id);
    } catch {
      warnings.push(`Could not fetch full text for ${candidate.title}.`);
    }

    try {
      const explanation = await explainMatch(caseProfile, candidate);
      matches.push({
        id: candidate.id,
        source: candidate.source,
        external_id: candidate.external_id,
        title: candidate.title,
        court: candidate.court,
        date: candidate.date,
        url: candidate.url,
        snippet: candidate.snippet,
        score: rankedCandidate.score,
        score_breakdown: rankedCandidate.score_breakdown,
        matching_points: explanation.matching_points,
        differences: explanation.differences,
        important_points: explanation.important_points,
        reusable_arguments: explanation.reusable_arguments,
        confidence: explanation.confidence,
        attribution: attributionFor(candidate.source),
      });
    } catch {
      matches.push({
        id: candidate.id,
        source: candidate.source,
        external_id: candidate.external_id,
        title: candidate.title,
        court: candidate.court,
        date: candidate.date,
        url: candidate.url,
        snippet: candidate.snippet,
        score: rankedCandidate.score,
        score_breakdown: rankedCandidate.score_breakdown,
        matching_points: [],
        differences: [],
        important_points: [],
        reusable_arguments: [],
        confidence: 'low',
        attribution: attributionFor(candidate.source),
      });
      warnings.push(`AI explanation unavailable for ${candidate.title}.`);
    }
  }

  return {
    case_profile: caseProfile,
    queries: generatedQueries,
    matches,
    warnings,
    generated_at: now().toISOString(),
  };
}

import type { NextApiRequest, NextApiResponse } from 'next';
import formidable from 'formidable';
import fs from 'fs';
import { validateFileSize, validateFileType } from '@/lib/pdf/validator';
import { validatePdf } from '@/lib/pdf/validator';
import { extractPdfText } from '@/lib/pdf/extractor';
import { cleanExtractedText, truncateForAI } from '@/lib/pdf/cleaner';
import { getErrorMessage } from '@/lib/pdf/errors';
import { runPrecedentResearch } from '@/lib/research/orchestrator';
import { IndianKanoonConnector } from '@/lib/research/sources/indianKanoon';

export const config = {
  api: {
    bodyParser: false,
  },
};

function getUploadedFiles(files: formidable.Files): formidable.File[] {
  const value = files.files || files.file || [];
  return Array.isArray(value) ? value : [value];
}

export default async function handler(req: NextApiRequest, res: NextApiResponse) {
  if (req.method !== 'POST') {
    return res.status(405).json({ success: false, message: 'Method not allowed' });
  }

  const form = formidable({
    maxFileSize: 10 * 1024 * 1024,
    keepExtensions: true,
    multiples: true,
  });

  let parsedFiles: formidable.File[] = [];

  try {
    const [, files] = await form.parse(req);
    parsedFiles = getUploadedFiles(files);

    if (parsedFiles.length === 0) {
      return res.status(400).json({
        success: false,
        error: 'NO_FILE',
        message: 'Upload at least one PDF document.',
        retryable: false,
      });
    }

    const documents = [];

    for (const file of parsedFiles) {
      const typeValidation = validateFileType(file.mimetype || '');
      if (!typeValidation.valid) {
        return res.status(400).json({
          success: false,
          error: typeValidation.error,
          message: getErrorMessage(typeValidation.error!),
          retryable: false,
        });
      }

      const sizeValidation = validateFileSize(file.size, 10 * 1024 * 1024);
      if (!sizeValidation.valid) {
        return res.status(400).json({
          success: false,
          error: sizeValidation.error,
          message: getErrorMessage(sizeValidation.error!),
          retryable: false,
        });
      }

      const buffer = fs.readFileSync(file.filepath);
      const pdfValidation = validatePdf(buffer);
      if (!pdfValidation.valid) {
        return res.status(400).json({
          success: false,
          error: pdfValidation.error,
          message: getErrorMessage(pdfValidation.error!),
          retryable: false,
        });
      }

      const extraction = await extractPdfText(buffer);
      if (!extraction.success) {
        return res.status(422).json({
          success: false,
          error: extraction.error,
          message: getErrorMessage(extraction.error!),
          retryable: false,
        });
      }

      documents.push({
        name: file.originalFilename || 'uploaded-document.pdf',
        text: truncateForAI(cleanExtractedText(extraction.text!)),
      });
    }

    const data = await runPrecedentResearch({
      documents,
      connectors: [new IndianKanoonConnector()],
    });

    return res.status(200).json({ success: true, data });
  } catch (error: any) {
    console.error('Research API error:', error);
    return res.status(500).json({
      success: false,
      error: 'RESEARCH_FAILED',
      message: error.message || 'Precedent research failed. Please try again.',
      retryable: true,
    });
  } finally {
    for (const file of parsedFiles) {
      if (file.filepath && fs.existsSync(file.filepath)) {
        fs.unlinkSync(file.filepath);
      }
    }
  }
}

interface MultiFileUploadProps {

  onFilesChange: (files: File[]) => void;

export function MultiFileUpload({
  files,
  onFilesChange,
  maxSize = 10 * 1024 * 1024,
  className,
}: MultiFileUploadProps) {
  const [isDragging, setIsDragging] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const addFiles = (incoming: FileList | File[]) => {
    const nextFiles = Array.from(incoming);
    const invalid = nextFiles.find(file => file.type !== 'application/pdf');
    if (invalid) {
      setError('Only PDF files are accepted');
      return;
    }

    const oversized = nextFiles.find(file => file.size > maxSize);
    if (oversized) {
      setError('Each file must be 10MB or smaller');
      return;
    }

    setError(null);
    onFilesChange([...files, ...nextFiles]);
  };

  const removeFile = (index: number) => {
    onFilesChange(files.filter((_, currentIndex) => currentIndex !== index));
  };

  return (
    <div className={cn('w-full', className)}>
      <div
        className={cn(
          'border-2 border-dashed rounded-card p-4 sm:p-8 bg-background transition-all duration-200',
          isDragging ? 'border-primary bg-card' : 'border-border hover:border-accent hover:bg-card/50',
          error && 'border-red-500',
          files.length > 0 && 'border-accent bg-card'
        )}
        onDragEnter={event => {
          event.preventDefault();
          setIsDragging(true);
        }}
        onDragLeave={event => {
          event.preventDefault();
          setIsDragging(false);
        }}
        onDragOver={event => event.preventDefault()}
        onDrop={event => {
          event.preventDefault();
          setIsDragging(false);
          addFiles(event.dataTransfer.files);
        }}
      >
        <input
          id="research-file-upload"
          type="file"
          accept="application/pdf"
          multiple
          onChange={event => {
            if (event.target.files) addFiles(event.target.files);
            event.target.value = '';
          }}
          className="hidden"
        />

        <label htmlFor="research-file-upload" className="cursor-pointer flex flex-col items-center gap-3">

import React, { useEffect, useState } from 'react';
import { useRouter } from 'next/router';
import { Button } from '@/components/ui/Button';
import { EmptyState } from '@/components/ui/EmptyState';
import { Loader } from '@/components/ui/Loader';
import { ResearchDashboard } from '@/components/research/ResearchDashboard';
import { ResearchResult, ResearchResultSchema } from '@/lib/research/types';

export default function ResearchResultsPage() {
  const router = useRouter();
  const [result, setResult] = useState<ResearchResult | null>(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    const stored = localStorage.getItem('nyaysetu_research_result');
    if (stored) {
      try {
        setResult(ResearchResultSchema.parse(JSON.parse(stored)));
      } catch {
        setResult(null);
      }
    }
    setLoading(false);
  }, []);

  if (loading) {
    return (
      <div className="min-h-screen bg-background flex items-center justify-center">
        <Loader size="lg" color="accent" />
      </div>
    );
  }

  if (!result) {
    return (
      <div className="min-h-screen bg-background flex items-center justify-center px-4">
        <EmptyState onUpload={() => router.push('/research')} />
      </div>
    );
  }

  return (
    <div className="min-h-screen bg-background">
      <header className="border-b border-border bg-card">
        <div className="max-w-container mx-auto px-4 sm:px-6 py-3 sm:py-4 flex items-center justify-between">
          <h1 className="text-lg sm:text-xl font-serif font-semibold text-primary">Public Precedent Matches</h1>
